# Phase 05 - Layout & Navigation

## Objectif

Créer le layout principal avec TabView (menu en bas) et la navigation.

## Étapes

### 1. Créer AppLayoutComponent

Créer `src/app/layout/app-layout.component.ts`:

```typescript
import { Component } from '@angular/core';
import { RouterModule } from '@angular/router';
import { TabViewModule } from 'primeng/tabview';
import { AuthService } from '@core/services/auth.service';

@Component({
  selector: 'app-layout',
  standalone: true,
  imports: [RouterModule, TabViewModule],
  template: `
    <div class="app-container">
      <div class="content">
        <router-outlet></router-outlet>
      </div>

      <p-tabView [scrollable]="true" scrollHeight="400px" class="tab-bar">
        <p-tabPanel header="Match" leftIcon="pi pi-home">
          <app-match-detail></app-match-detail>
        </p-tabPanel>

        <p-tabPanel header="Événements" leftIcon="pi pi-list">
          <app-event-list></app-event-list>
        </p-tabPanel>

        <p-tabPanel header="Stats" leftIcon="pi pi-chart-bar">
          <app-match-stats></app-match-stats>
        </p-tabPanel>

        <p-tabPanel header="Plus" leftIcon="pi pi-ellipsis-h">
          <app-more-menu></app-more-menu>
        </p-tabPanel>
      </p-tabView>
    </div>
  `,
  styleUrls: ['./app-layout.component.scss']
})
export class AppLayoutComponent {
  constructor(private auth: AuthService) {}
}
```

### 2. Styler la TabView

Créer `src/app/layout/app-layout.component.scss`:

```scss
.app-container {
```

### 3. Configurer les routes

Éditer `src/app/app.routes.ts`:

```typescript
import { Routes } from '@angular/router';
import { AppLayoutComponent } from './layout/app-layout.component';
import { authGuard } from './core/guards/auth.guard';

export const routes: Routes = [
  { path: '', redirectTo: '/app/teams', pathMatch: 'full' },
  
  {
    path: 'auth',
    loadChildren: () => import('./features/auth/auth.routes').then(m => m.AUTH_ROUTES)
  },
  
  {
    path: 'app',
    component: AppLayoutComponent,
    canActivate: [authGuard],
    children: [
      { path: '', redirectTo: 'teams', pathMatch: 'full' },
      {
        path: 'teams',
        loadChildren: () => import('./features/teams/teams.routes').then(m => m.TEAMS_ROUTES)
      },
      {
        path: 'matches/:matchId',
        loadChildren: () => import('./features/matches/matches.routes').then(m => m.MATCHES_ROUTES)
      },
      {
        path: 'events',
        loadChildren: () => import('./features/events/events.routes').then(m => m.EVENTS_ROUTES)
      },
      {
        path: 'menu',
        loadChildren: () => import('./features/menu/menu.routes').then(m => m.MENU_ROUTES)
      }
    ]
  },
  
  { path: '**', redirectTo: '/app/teams' }
];
```

### 4. Créer les routes features

**teams.routes.ts:**
```typescript
import { Routes } from '@angular/router';

export const TEAMS_ROUTES: Routes = [
  { path: '', component: TeamListComponent },
  { path: 'new', component: TeamFormComponent },
  { path: ':teamId', component: TeamDetailComponent },
  { path: ':teamId/edit', component: TeamFormComponent }
];
```

Même pattern pour matches.routes.ts, events.routes.ts, menu.routes.ts
