# Phase 07 - Matches Feature

## Objectif

Implémenter la gestion des matchs avec timer et affichage du score.

## Étapes

### 1. Créer MatchService

Créer `src/app/features/matches/services/match.service.ts`:

```typescript
import { Injectable } from '@angular/core';
import { BehaviorSubject, Observable, interval } from 'rxjs';
import { map } from 'rxjs/operators';
import { Match, Evenement } from '@core/models/datamodel';
import { DatabaseService } from '@core/services/database.service';

@Injectable({
  providedIn: 'root'
})
export class MatchService {
  private currentMatch$ = new BehaviorSubject<Match | null>(null);
  private matchTimer$ = new BehaviorSubject<{ minutes: number, seconds: number }>({ minutes: 0, seconds: 0 });

  constructor(private db: DatabaseService) {}

  async addMatch(match: Omit<Match, 'id' | 'createdAt' | 'updatedAt'>): Promise<Match> {
    const newMatch: Omit<Match, 'id'> = {
      ...match,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };
    return this.db.addMatch(newMatch);
  }

  async getMatch(matchId: number): Promise<Match | undefined> {
    const match = await this.db.getMatch(matchId);
    if (match) {
      this.currentMatch$.next(match);
    }
    return match;
  }

  getCurrentMatch(): Observable<Match | null> {
    return this.currentMatch$.asObservable();
  }

  getMatchTimer(): Observable<{ minutes: number, seconds: number }> {
    return this.matchTimer$.asObservable();
  }

  startTimer() {
    let seconds = 0;
    interval(1000).subscribe(() => {
      seconds++;
      const minutes = Math.floor(seconds / 60);
      const secs = seconds % 60;
      this.matchTimer$.next({ minutes, seconds: secs });
    });
  }

  async calculateScore(matchId: number): Promise<{ nous: number, adversaire: number }> {
    const events = await this.db.getEventsByMatch(matchId);
    const scoreEvents = events.filter(e => e.nature === 'SCORE');
    
    const pointsMap: any = {
      'ESSAI': { 'MARQUE': 5, 'RATE': 0 },
      'TRANSFORMATION': { 'REUSSITE': 2, 'ECHEC': 0 },
      'PENALITE': { 'MARQUE': 3, 'RATE': 0 },
      'DROP': { 'MARQUE': 3, 'RATE': 0 }
    };

    let nous = 0, adversaire = 0;

    scoreEvents.forEach(e => {
      const points = pointsMap[e.type]?.[e.resultat] || 0;
      if (e.equipe === 'NOUS') nous += points;
      else adversaire += points;
    });

    return { nous, adversaire };
  }
}
```

### 2. Créer MatchDetailComponent

Créer `src/app/features/matches/pages/match-detail/match-detail.component.ts`:

```typescript
import { Component, OnInit } from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import { CommonModule } from '@angular/common';
import { Match, Evenement } from '@core/models/datamodel';
import { MatchService } from '../../services/match.service';
import { EventService } from '@features/events/services/event.service';
import { CardModule } from 'primeng/card';
import { ButtonModule } from 'primeng/button';
import { Observable } from 'rxjs';

@Component({
  selector: 'app-match-detail',
  standalone: true,
  imports: [CommonModule, CardModule, ButtonModule],
  template: `
    <div *ngIf="match$ | async as match" class="match-detail">
      <div class="score-display">
        <div class="team-nous">{{ match.score.nous }}</div>
        <div class="match-info">
          {{ currentTimer$ | async | json }}
        </div>
        <div class="team-adv">{{ match.score.adversaire }}</div>
      </div>

      <h2>Événements</h2>
      <div class="events-list">
        <div *ngFor="let event of (events$ | async) as events"
             class="event-item">
          {{ event.instant }} - {{ event.type }} ({{ event.equipe }})
          <p-button 
            icon="pi pi-pencil" 
            [text]="true"
            (click)="editEvent(event)">
          </p-button>
          <p-button 
            icon="pi pi-trash" 
            [text]="true"
            severity="danger"
            (click)="deleteEvent(event)">
          </p-button>
        </div>
      </div>

      <p-button 
        icon="pi pi-plus"
        class="fab"
        (click)="createEvent()">
      </p-button>
    </div>
  `,
  styleUrls: ['./match-detail.component.scss']
})
export class MatchDetailComponent implements OnInit {
  match$!: Observable<Match | null>;
  events$!: Observable<Evenement[]>;
  currentTimer$!: Observable<{ minutes: number, seconds: number }>;

  constructor(
    private route: ActivatedRoute,
    private router: Router,
    private matchService: MatchService,
    private eventService: EventService
  ) {}

  ngOnInit() {
    const matchId = this.route.snapshot.paramMap.get('matchId');
    if (matchId) {
      this.matchService.getMatch(+matchId);
      this.match$ = this.matchService.getCurrentMatch();
      this.events$ = this.eventService.getEventsByMatchAntiChronological(+matchId);
      this.currentTimer$ = this.matchService.getMatchTimer();
      this.matchService.startTimer();
    }
  }

  createEvent() {
    const matchId = this.route.snapshot.paramMap.get('matchId');
    this.router.navigate(['/app/events/new'], { queryParams: { matchId } });
  }

  editEvent(event: Evenement) {
    this.router.navigate(['/app/events', event.id, 'edit']);
  }

  async deleteEvent(event: Evenement) {
    if (confirm('Supprimer cet événement?')) {
      const matchId = this.route.snapshot.paramMap.get('matchId');
      if (matchId) {
        await this.eventService.deleteEvent(+matchId, event.id);
      }
    }
  }
}
```

### 3. Créer MatchFormComponent

Créer `src/app/features/matches/pages/match-form/match-form.component.ts`:

```typescript
import { Component, OnInit } from '@angular/core';
import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { ActivatedRoute, Router } from '@angular/router';
import { DialogModule } from 'primeng/dialog';
import { ButtonModule } from 'primeng/button';
import { InputTextModule } from 'primeng/inputtext';
import { CalendarModule } from 'primeng/calendar';
import { RadioButtonModule } from 'primeng/radiobutton';
import { DropdownModule } from 'primeng/dropdown';
import { MatchService } from '../../services/match.service';

@Component({
  selector: 'app-match-form',
  standalone: true,
  imports: [
    ReactiveFormsModule, DialogModule, ButtonModule,
    InputTextModule, CalendarModule, RadioButtonModule, DropdownModule
  ],
  template: `
    <p-dialog 
      [(visible)]="displayDialog" 
      [modal]="true"
      header="Nouveau Match">
      
      <form [formGroup]="form">
        <div class="form-group">
          <label>Adversaire</label>
          <input pInputText formControlName="nomAdversaire">
        </div>

        <div class="form-group">
          <label>Date</label>
          <p-calendar formControlName="date" [showIcon]="true"></p-calendar>
        </div>

        <div class="form-group">
          <label>Terrain</label>
          <div>
            <p-radioButton 
              name="terrain" 
              value="NATUREL"
              formControlName="terrain"
              label="Naturel">
            </p-radioButton>
            <p-radioButton 
              name="terrain" 
              value="SYNTHETIQUE"
              formControlName="terrain"
              label="Synthétique">
            </p-radioButton>
          </div>
        </div>

        <div class="form-group">
          <label>Conditions météo</label>
          <p-dropdown formControlName="conditions"
                      [options]="weatherOptions"
                      optionLabel="label"
                      optionValue="value">
          </p-dropdown>
        </div>
      </form>

      <ng-template pTemplate="footer">
        <p-button label="Annuler" (click)="cancel()"></p-button>
        <p-button label="Créer" (click)="save()"></p-button>
      </ng-template>
    </p-dialog>
  `
})
export class MatchFormComponent implements OnInit {
  form: FormGroup;
  displayDialog = true;
  weatherOptions = [
    { label: 'Sec', value: 'SEC' },
    { label: 'Pluie', value: 'PLUIE' },
    { label: 'Vent', value: 'VENT' }
  ];

  constructor(
    private fb: FormBuilder,
    private matchService: MatchService,
    private route: ActivatedRoute,
    private router: Router
  ) {
    this.form = this.fb.group({
      nomAdversaire: ['', Validators.required],
      date: ['', Validators.required],
      terrain: ['NATUREL', Validators.required],
      conditions: ['SEC']
    });
  }

  ngOnInit() {}

  async save() {
    if (!this.form.valid) return;

    const teamId = this.route.snapshot.queryParamMap.get('teamId');
    if (!teamId) return;

    try {
      await this.matchService.addMatch({
        equipeId: +teamId,
        ...this.form.value,
        score: { nous: 0, adversaire: 0 }
      });
      this.router.navigate(['/app/teams', teamId]);
    } catch (error) {
      console.error('Error creating match:', error);
    }
  }

  cancel() {
    this.router.back();
  }
}
```
