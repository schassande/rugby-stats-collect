# Phase 09b - Réauthentification & Sync Local→Cloud

## Contexte

Quand un utilisateur s'est connecté en mode **local** (offline), ses données sont dans IndexedDB. S'il reconnecte à Internet et veut synchroniser avec Firestore, il doit d'abord se réauthentifier avec Firebase.

## Architecture

### Scénario

```
T0: Utilisateur connecté mode LOCAL
    └─ 3 événements créés localement
    └─ operations_queue: 3 "pending"
    └─ localStorage: auth_mode='local'

T1: Internet restauré
    └─ App détecte: isOnline = true

T2: SyncService tente de syncer
    └─ Vérification: currentUser (Firebase) = NULL
    └─ Mais operations_queue a des ops!
    └─ Conflit: operations pending + no firebase user

T3: Modal "Réauthentification requise"
    └─ Message: "Pour synchroniser vos données, authentifiez-vous Firebase"
    └─ Champs: Email + Mot de passe (Firebase)

T4: Utilisateur saisit credentials
    └─ Vérif Firebase Auth
    └─ Si succès: Token généré

T5: SyncService relance
    └─ Maintenant: + Firebase token
    └─ 3 événements uploadés
    └─ Operations marquées "synced"

T6: UI: "3 événements synchronisés ✓"
```

### 2. Créer ReauthService

Créer `src/app/core/services/reauth.service.ts`:

```typescript
import { Injectable, inject } from '@angular/core';
import { DialogService, DynamicDialogRef } from 'primeng/dynamicdialog';
import { ReauthModalComponent } from '@features/auth/pages/reauth-modal/reauth-modal.component';
import { signInWithEmailAndPassword, Auth } from 'firebase/auth';
import { auth } from '../config/firebase.config';

@Injectable({
  providedIn: 'root'
})
export class ReauthService {
  private dialogService = inject(DialogService);
  private dialogRef: DynamicDialogRef | null = null;

  /**
   * Demander réauthentification utilisateur
   * Retourne: Firebase token si succès, null si cancelled
   */
  async promptReauth(): Promise<string | null> {
    return new Promise((resolve) => {
      this.dialogRef = this.dialogService.open(ReauthModalComponent, {
        header: 'Authentification requise',
        modal: true,
        closable: false,
        resizable: false
      });

      this.dialogRef.onClose.subscribe(async (credentials) => {
        if (!credentials) {
          resolve(null);
          return;
        }

        try {
          // Vérifier avec Firebase
          const userCredential = await signInWithEmailAndPassword(
            auth,
            credentials.email,
            credentials.password
          );

          // Récupérer token
          const token = await userCredential.user.getIdToken();
          resolve(token);
        } catch (error: any) {
          // Erreur auth - relancer modal
          console.error('Reauth failed:', error);
          const retryToken = await this.promptReauth();
          resolve(retryToken);
        }
      });
    });
  }

  /**
   * Vérifier si réauth nécessaire
   * Retourne true si:
   * - Mode local ET
   * - Pas de Firebase user AND
   * - Operations pending existent
   */
  async isReauthRequired(
    authMode: 'local' | 'firebase',
    hasFirebaseUser: boolean,
    pendingOpsCount: number
  ): Promise<boolean> {
    return authMode === 'local' && !hasFirebaseUser && pendingOpsCount > 0;
  }
}
```

### 3. Modifier SyncService

Éditer `src/app/core/services/sync.service.ts`:

```typescript
import { Injectable, NgZone } from '@angular/core';
import { BehaviorSubject, Observable, interval, fromEvent } from 'rxjs';
import { filter, switchMap } from 'rxjs/operators';
import { AuthService } from './auth.service';
import { ReauthService } from './reauth.service';
import { QueueService } from './queue.service';
import { FirestoreService } from './firestore.service';
import { environment } from '@env/environment';

@Injectable({
  providedIn: 'root'
})
export class SyncService {
  private isOnlineSubject = new BehaviorSubject<boolean>(navigator.onLine);
  isOnline$ = this.isOnlineSubject.asObservable();

  private isSyncingSubject = new BehaviorSubject<boolean>(false);
  isSyncing$ = this.isSyncingSubject.asObservable();

  private pendingCountSubject = new BehaviorSubject<number>(0);
  pendingCount$ = this.pendingCountSubject.asObservable();

  private syncErrorSubject = new BehaviorSubject<string | null>(null);
  syncError$ = this.syncErrorSubject.asObservable();

  constructor(
    private auth: AuthService,
    private reauth: ReauthService,
    private queue: QueueService,
    private firestore: FirestoreService,
    private ngZone: NgZone
  ) {
    this.setupConnectivityListener();
    this.setupAutoSync();
  }

  private setupConnectivityListener() {
    this.ngZone.runOutsideAngular(() => {
      fromEvent(window, 'online')
        .pipe(filter(() => true))
        .subscribe(() => {
          this.ngZone.run(() => {
            this.isOnlineSubject.next(true);
            // Trigger sync immédiatement
            this.startSync();
          });
        });

      fromEvent(window, 'offline')
        .pipe(filter(() => true))
        .subscribe(() => {
          this.ngZone.run(() => this.isOnlineSubject.next(false));
        });
    });
  }

  private setupAutoSync() {
    this.ngZone.runOutsideAngular(() => {
      interval(5000) // Check every 5s
        .pipe(
          switchMap(() => this.isOnline$),
          filter(isOnline => isOnline)
        )
        .subscribe(() => {
          this.ngZone.run(() => this.startSync());
        });
    });
  }

  async startSync() {
    if (this.isSyncingSubject.value) return;

    this.isSyncingSubject.next(true);
    try {
      const operations = await this.queue.getPendingOperations();
      this.pendingCountSubject.next(operations.length);

      if (operations.length === 0) {
        return; // Rien à syncer
      }

      // ========== NOUVEAU: Vérifier réauth ==========
      const authMode = environment.authMode as 'local' | 'firebase';
      const currentUser = this.auth.getCurrentUser();
      
      const needsReauth = await this.reauth.isReauthRequired(
        authMode,
        !!currentUser,
        operations.length
      );

      if (needsReauth) {
        // Demander réauthentification
        const token = await this.reauth.promptReauth();
        
        if (!token) {
          // User cancelled
          this.syncErrorSubject.next('Réauthentification annulée');
          return;
        }

        // Sauver token pour les requêtes
        localStorage.setItem('firebase_token', token);
        localStorage.setItem('auth_mode', 'firebase');
      }
      // ===== FIN NOUVEAU =====

      // Group by matchId for batching
      const grouped = new Map<number, any[]>();
      operations.forEach(op => {
        if (!grouped.has(op.matchId)) grouped.set(op.matchId, []);
        grouped.get(op.matchId)!.push(op);
      });

      // Sync each group
      for (const [matchId, ops] of grouped.entries()) {
        for (const op of ops) {
          await this.syncOperation(op);
        }
      }

      this.syncErrorSubject.next(null);
    } catch (error: any) {
      this.syncErrorSubject.next(error.message);
    } finally {
      this.isSyncingSubject.next(false);
      const pending = await this.queue.getPendingOperations();
      this.pendingCountSubject.next(pending.length);
    }
  }

  private async syncOperation(operation: any) {
    await this.queue.updateOperationStatus(operation.id, 'syncing');

    try {
      const maxRetries = 5;
      const baseDelay = 1000;

      for (let attempt = 0; attempt < maxRetries; attempt++) {
        try {
          if (operation.operation === 'create') {
            await this.firestore.createEvent(operation.data);
          } else if (operation.operation === 'update') {
            await this.firestore.updateEvent(operation.data);
          } else if (operation.operation === 'delete') {
            await this.firestore.deleteEvent(operation.data.id);
          }

          await this.queue.updateOperationStatus(operation.id, 'synced', 0);
          return;
        } catch (error: any) {
          if (attempt < maxRetries - 1 && error.code !== 'permission-denied') {
            const delay = baseDelay * Math.pow(2, attempt);
            await new Promise(resolve => setTimeout(resolve, delay));
          } else if (attempt === maxRetries - 1) {
            throw error;
          }
        }
      }
    } catch (error: any) {
      await this.queue.updateOperationStatus(
        operation.id,
        'failed',
        operation.retryCount + 1
      );
      throw error;
    }
  }
}
```

## Flux Complet (Exemple)

```typescript
// Utilisateur connecté mode LOCAL offline
await auth.signInLocal('test@rugby.local', 'password123');
// ✅ localStorage: auth_mode='local'

// Créer 3 événements
await eventService.addEvent({type: 'ESSAI', resultat: 'MARQUE'});
await eventService.addEvent({type: 'TRANSFORMATION', resultat: 'REUSSITE'});
await eventService.addEvent({type: 'PENALITE', resultat: 'RATE'});
// ✅ IndexedDB: 3 opérations "pending"

// Passer offline → online
window.dispatchEvent(new Event('online'));

// SyncService détecte et tente sync
// → Vérif: auth_mode='local' + no firebase user + 3 pending ops
// → Modal: "Réauthentification requise"
// User: Saisit 'john@gmail.com' / 'firebase_password'
// Firebase Auth OK → Token généré
// → SyncService relance avec token
// → 3 événements uploadés à Firestore
// → localStorage: auth_mode='firebase'
// → UI: "3 événements synchronisés ✓"

// À partir de là: synchronisation normale
```

## Limitations

### Avant Réauth
- ❌ Pas d'accès Firestore
- ❌ Pas de sync cloud
- ✅ Mais: Toutes les données locales accessibles

### Après Réauth
- ✅ Accès Firestore
- ✅ Sync automatique
- ✅ Données local + cloud synchronisées
