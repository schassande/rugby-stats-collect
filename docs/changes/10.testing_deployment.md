### 1. Tests Unitaires

Créer `src/app/core/services/queue.service.spec.ts`:

```typescript
import { TestBed } from '@angular/core/testing';
import { QueueService } from './queue.service';
import { db } from '../database/rugby-stats.database';

describe('QueueService', () => {
  let service: QueueService;

  beforeEach(async () => {
    TestBed.configureTestingModule({});
    service = TestBed.inject(QueueService);
    await db.delete(); // Clear DB before each test
  });

  it('should add operation to queue', async () => {
    const operation = {
      id: '1',
      evenementId: 1,
      matchId: 1,
      operation: 'create' as const,
      status: 'pending' as const,
      data: {} as any,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      retryCount: 0
    };

    await service.addOperation(operation);
    const pending = await service.getPendingOperations();
    expect(pending.length).toBe(1);
  });

  it('should update operation status', async () => {
    const operation = {
      id: '1',
      evenementId: 1,
      matchId: 1,
      operation: 'create' as const,
      status: 'pending' as const,
      data: {} as any,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      retryCount: 0
    };

    await service.addOperation(operation);
    await service.updateOperationStatus('1', 'synced');
    const synced = await service.getSyncedOperations();
    expect(synced.length).toBe(1);
  });
});
```

### 2. Tests d'Intégration Offline

Créer `src/app/core/services/sync.service.spec.ts`:

```typescript
import { TestBed } from '@angular/core/testing';
import { SyncService } from './sync.service';

describe('SyncService Offline', () => {
  let service: SyncService;

  beforeEach(() => {
    TestBed.configureTestingModule({
      providers: [SyncService]
    });
    service = TestBed.inject(SyncService);
  });

  it('should detect online status', (done) => {
    service.isOnline$.subscribe(isOnline => {
      expect(typeof isOnline).toBe('boolean');
      done();
    });
  });

  it('should handle offline scenario', async () => {
    // Simulate going offline
    Object.defineProperty(window.navigator, 'onLine', {
      writable: true,
      value: false
    });

    window.dispatchEvent(new Event('offline'));

    service.isOnline$.subscribe(isOnline => {
      expect(isOnline).toBe(false);
    });
  });

  it('should auto-sync when going online', async () => {
    // Start offline
    Object.defineProperty(window.navigator, 'onLine', {
      writable: true,
      value: false
    });
    window.dispatchEvent(new Event('offline'));

    // Go online
    Object.defineProperty(window.navigator, 'onLine', {
      writable: true,
      value: true
    });
    window.dispatchEvent(new Event('online'));

    // Should trigger sync
    service.isSyncing$.subscribe(isSyncing => {
      // Verify sync started
    });
  });
});
```

### 3. Tester Offline en Dev

```bash
# Build production
ng build --configuration production

# Serve
ng serve --poll 2000

# DevTools:
# 1. Application → Service Workers
# 2. Network → select "Offline"
# 3. Ajouter des événements
# 4. Vérifier IndexedDB (Application → IndexedDB)
# 5. Voir operations_queue en "pending"
# 6. Passer en ligne → Auto-sync
# 7. Voir status passer à "synced"
```

### 4. Tester sur Device Physique

```bash
# Build pour production
ng build --configuration production

# Déployer sur Firebase Hosting (preview)
firebase hosting:channel:deploy preview-channel --build-dir=dist/games-stats

# Sur mobile:
# 1. Ouvrir l'URL preview
# 2. Menu Chrome → "Installer l'app"
# 3. Déactiver WiFi/données
# 4. Utiliser l'app offline
# 5. Voir les données se sauvegarder localement
# 6. Réactiver WiFi
# 7. Vérifier sync automatique
```

### 5. Déployer sur Firebase Hosting

```bash
# 1. Login
firebase login

# 2. Initialiser Firebase Hosting
firebase init hosting

# 3. Select "dist/games-stats" as public directory

# 4. Build production
ng build --configuration production

# 5. Deploy
firebase deploy --only hosting

# 6. Vérifier à https://your-project.web.app
```
