# Phase 02 - Firebase Authentication

## Objectif

Configurer Firebase Authentication (Email/Password + Google Sign-In).

## Prérequis

- Projet Angular créé (Phase 01)
- Compte Firebase

## Étapes

### 1. Créer un projet Firebase

1. Aller sur [Firebase Console](https://console.firebase.google.com/)
2. Créer un nouveau projet
3. Ajouter une application Web
4. Copier la configuration (firebaseConfig)

### 3. Implémenter AuthService

Créer `src/app/core/services/auth.service.ts`:

```typescript
import { Injectable } from '@angular/core';
import { Auth, signInWithEmailAndPassword, createUserWithEmailAndPassword, 
         signOut, signInWithPopup, GoogleAuthProvider, onAuthStateChanged, User } from 'firebase/auth';
import { BehaviorSubject, Observable, from } from 'rxjs';
import { auth } from '../config/firebase.config';
import { Firestore, collection, doc, setDoc } from 'firebase/firestore';
import { db } from '../config/firebase.config';

export interface Manager {
  id: string;
  prenom: string;
  nom: string;
  email: string;
  createdAt: string;
  updatedAt: string;
}

@Injectable({
  providedIn: 'root'
})
export class AuthService {
  private currentUserSubject = new BehaviorSubject<User | null>(null);
  public currentUser$ = this.currentUserSubject.asObservable();

  private isAuthenticatedSubject = new BehaviorSubject<boolean>(false);
  public isAuthenticated$ = this.isAuthenticatedSubject.asObservable();

  constructor() {
    this.setupAuthListener();
  }

  private setupAuthListener() {
    onAuthStateChanged(auth, (user) => {
      this.currentUserSubject.next(user);
      this.isAuthenticatedSubject.next(!!user);
    });
  }

  // Email/Password signup
  async signUpWithEmail(email: string, password: string, prenom: string, nom: string): Promise<void> {
    try {
      const userCredential = await createUserWithEmailAndPassword(auth, email, password);
      const user = userCredential.user;

      // Créer document Manager dans Firestore
      const manager: Manager = {
        id: user.uid,
        prenom,
        nom,
        email,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString()
      };

      await setDoc(doc(db, 'managers', user.uid), manager);
    } catch (error) {
      throw error;
    }
  }

  // Email/Password signin
  async signInWithEmail(email: string, password: string): Promise<void> {
    try {
      await signInWithEmailAndPassword(auth, email, password);
    } catch (error) {
      throw error;
    }
  }

  // Google Sign-In
  async signInWithGoogle(): Promise<void> {
    try {
      const provider = new GoogleAuthProvider();
      const userCredential = await signInWithPopup(auth, provider);
      const user = userCredential.user;

      // Créer document Manager si première connexion
      const manager: Manager = {
        id: user.uid,
        prenom: user.displayName?.split(' ')[0] || '',
        nom: user.displayName?.split(' ')[1] || '',
        email: user.email || '',
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString()
      };

      await setDoc(doc(db, 'managers', user.uid), manager, { merge: true });
    } catch (error) {
      throw error;
    }
  }

  // Sign out
  async signOut(): Promise<void> {
    try {
      await signOut(auth);
    } catch (error) {
      throw error;
    }
  }

  // Getters
  getCurrentUser(): User | null {
    return this.currentUserSubject.value;
  }

  isAuthenticated(): boolean {
    return this.isAuthenticatedSubject.value;
  }

  getIdToken(): Promise<string | null> {
    const user = this.getCurrentUser();
    if (!user) return Promise.resolve(null);
    return user.getIdToken();
  }
}
```

### 4. Créer LoginComponent

Créer `src/app/features/auth/pages/login/login.component.ts`:

```typescript
import { Component } from '@angular/core';
import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { Router } from '@angular/router';
import { AuthService } from '@core/services/auth.service';
import { ButtonModule } from 'primeng/button';
import { InputTextModule } from 'primeng/inputtext';
import { MessageModule } from 'primeng/message';
import { CommonModule } from '@angular/common';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule, ButtonModule, InputTextModule, MessageModule],
  template: `
    <div class="login-container">
      <h1>Rugby Stats Collector</h1>
      
      <form [formGroup]="form">
        <div class="form-group">
          <label>Email</label>
          <input pInputText formControlName="email" type="email" placeholder="Email">
        </div>

        <div class="form-group">
          <label>Mot de passe</label>
          <input pInputText formControlName="password" type="password" placeholder="Mot de passe">
        </div>

        <p-button label="Connexion" (click)="login()" [loading]="isLoading"></p-button>
        <p-button label="Google Sign-In" icon="pi pi-google" (click)="loginWithGoogle()"></p-button>
      </form>

      <p-message *ngIf="error" severity="error" [text]="error"></p-message>

      <p>Pas encore de compte? <a routerLink="/auth/signup">Créer un compte</a></p>
    </div>
  `,
  styleUrls: ['./login.component.scss']
})
export class LoginComponent {
  form: FormGroup;
  isLoading = false;
  error: string | null = null;

  constructor(
    private fb: FormBuilder,
    private auth: AuthService,
    private router: Router
  ) {
    this.form = this.fb.group({
      email: ['', [Validators.required, Validators.email]],
      password: ['', [Validators.required, Validators.minLength(6)]]
    });
  }

  async login() {
    if (!this.form.valid) return;

    this.isLoading = true;
    try {
      await this.auth.signInWithEmail(this.form.value.email, this.form.value.password);
      this.router.navigate(['/app/teams']);
    } catch (error: any) {
      this.error = this.getErrorMessage(error.code);
    } finally {
      this.isLoading = false;
    }
  }

  async loginWithGoogle() {
    this.isLoading = true;
    try {
      await this.auth.signInWithGoogle();
      this.router.navigate(['/app/teams']);
    } catch (error: any) {
      this.error = this.getErrorMessage(error.code);
    } finally {
      this.isLoading = false;
    }
  }

  private getErrorMessage(code: string): string {
    const errors: { [key: string]: string } = {
      'auth/user-not-found': 'Email non trouvé',
      'auth/wrong-password': 'Mot de passe incorrect',
      'auth/invalid-email': 'Email invalide',
      'auth/user-disabled': 'Compte désactivé'
    };
    return errors[code] || 'Erreur d\'authentification';
  }
}
```

import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { Router } from '@angular/router';
import { AuthService } from '@core/services/auth.service';
import { ButtonModule } from 'primeng/button';
import { InputTextModule } from 'primeng/inputtext';
import { MessageModule } from 'primeng/message';
import { CheckboxModule } from 'primeng/checkbox';
import { CommonModule } from '@angular/common';

@Component({
  selector: 'app-signup',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule, ButtonModule, InputTextModule, MessageModule, CheckboxModule],
  template: `
    <div class="signup-container">
      <h1>Créer mon compte</h1>
      
      <form [formGroup]="form">
        <div class="form-group">
          <label>Prénom</label>
          <input pInputText formControlName="prenom" placeholder="Prénom">
        </div>

        <div class="form-group">
          <label>Nom</label>
          <input pInputText formControlName="nom" placeholder="Nom">
        </div>

        <div class="form-group">
          <label>Email</label>
          <input pInputText formControlName="email" type="email" placeholder="Email">
        </div>

        <div class="form-group">
          <label>Mot de passe</label>
          <input pInputText formControlName="password" type="password" placeholder="Min 6 caractères">
        </div>

        <div class="form-group">
          <label>Confirmer mot de passe</label>
          <input pInputText formControlName="confirmPassword" type="password" placeholder="Confirmer">
        </div>

        <div class="form-group">
          <p-checkbox [(ngModel)]="agreedToCGU" name="cgu"></p-checkbox>
          <label>J'accepte les conditions d'utilisation</label>
        </div>

        <p-button label="Créer mon compte" (click)="signup()" [loading]="isLoading"></p-button>
        <p-button label="Google Sign-In" icon="pi pi-google" (click)="signupWithGoogle()"></p-button>
      </form>

      <p-message *ngIf="error" severity="error" [text]="error"></p-message>

      <p>Déjà un compte? <a routerLink="/auth/login">Connexion</a></p>
    </div>
  `,
  styleUrls: ['./signup.component.scss']
})
export class SignupComponent {
  form: FormGroup;
  isLoading = false;
  error: string | null = null;
  agreedToCGU = false;

  constructor(
    private fb: FormBuilder,
    private auth: AuthService,
    private router: Router
  ) {
    this.form = this.fb.group({
      prenom: ['', Validators.required],
      nom: ['', Validators.required],
      email: ['', [Validators.required, Validators.email]],
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmPassword: ['', Validators.required]
    });
  }

  async signup() {
    if (!this.form.valid || !this.agreedToCGU) {
      this.error = 'Veuillez remplir tous les champs';
      return;
    }

    if (this.form.value.password !== this.form.value.confirmPassword) {
      this.error = 'Les mots de passe ne correspondent pas';
      return;
    }

    this.isLoading = true;
    try {
      await this.auth.signUpWithEmail(
        this.form.value.email,
        this.form.value.password,
        this.form.value.prenom,
        this.form.value.nom
      );
      this.router.navigate(['/app/teams']);
    } catch (error: any) {
      this.error = this.getErrorMessage(error.code);
    } finally {
      this.isLoading = false;
    }
  }

  async signupWithGoogle() {
    this.isLoading = true;
    try {
      await this.auth.signInWithGoogle();
      this.router.navigate(['/app/teams']);
    } catch (error: any) {
      this.error = this.getErrorMessage(error.code);
    } finally {
      this.isLoading = false;
    }
  }

  private getErrorMessage(code: string): string {
    const errors: { [key: string]: string } = {
      'auth/email-already-in-use': 'Email déjà utilisé',
      'auth/weak-password': 'Mot de passe trop faible',
      'auth/invalid-email': 'Email invalide'
    };
    return errors[code] || 'Erreur lors de la création du compte';
  }
}

### 6. Créer AuthGuard

Créer `src/app/core/guards/auth.guard.ts`:

```typescript
import { Injectable } from '@angular/core';
import { CanActivateFn, Router } from '@angular/router';
import { AuthService } from '../services/auth.service';
import { inject } from '@angular/core';

export const authGuard: CanActivateFn = (route, state) => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.isAuthenticated()) {
    return true;
  }

  router.navigate(['/auth/login'], { queryParams: { returnUrl: state.url } });
  return false;
};
```

### 7. Configurer Firestore Rules

Aller dans Firebase Console → Firestore Database → Rules:

```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /managers/{uid} {
      allow read, write: if request.auth.uid == uid;
    }
    
    match /teams/{teamId} {
      allow read: if request.auth.uid in resource.data.managerIds;
      allow write: if request.auth.uid in resource.data.managerIds;
      
      match /matches/{matchId} {
        allow read, write: if request.auth.uid in get(/databases/$(database)/documents/teams/$(teamId)).data.managerIds;
        
        match /evenements/{eventId} {
          allow read, write: if request.auth.uid in get(/databases/$(database)/documents/teams/$(teamId)).data.managerIds;
        }
      }
    }
  }
}
```
