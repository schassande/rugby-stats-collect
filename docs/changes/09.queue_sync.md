## Objectif

Implémenter la file d'attente locale et la synchronisation Firestore avec gestion des conflits.

## Étapes

### 1. Créer QueueService

Créer `src/app/core/services/queue.service.ts`:

```typescript
import { Injectable } from '@angular/core';
import { BehaviorSubject, Observable } from 'rxjs';
import { SyncOperation } from '../models/datamodel';
import { db } from '../database/rugby-stats.database';

@Injectable({
  providedIn: 'root'
})
export class QueueService {
  private queueSubject = new BehaviorSubject<SyncOperation[]>([]);
  queue$ = this.queueSubject.asObservable();

  constructor() {
    this.loadQueue();
  }

  private async loadQueue() {
    const operations = await db.operations_queue.toArray();
    this.queueSubject.next(operations);
  }

  async addOperation(operation: SyncOperation): Promise<void> {
    await db.operations_queue.add(operation);
    await this.loadQueue();
  }

  async updateOperationStatus(
    operationId: string, 
    status: 'pending' | 'syncing' | 'synced' | 'conflict' | 'failed',
    retryCount?: number
  ): Promise<void> {
    const op = await db.operations_queue.get(operationId);
    if (op) {
      op.status = status;
      op.updatedAt = new Date().toISOString();
      if (retryCount !== undefined) op.retryCount = retryCount;
      await db.operations_queue.put(op);
      await this.loadQueue();
    }
  }

  async removeOperation(operationId: string): Promise<void> {
    await db.operations_queue.delete(operationId);
    await this.loadQueue();
  }

  getPendingOperations(): Promise<SyncOperation[]> {
    return db.operations_queue.where('status').equals('pending').toArray();
  }

  getSyncedOperations(): Promise<SyncOperation[]> {
    return db.operations_queue.where('status').equals('synced').toArray();
  }

  getFailedOperations(): Promise<SyncOperation[]> {
    return db.operations_queue.where('status').equals('failed').toArray();
  }

  async clearQueue(): Promise<void> {
    await db.operations_queue.clear();
    await this.loadQueue();
  }
}
```

### 2. Créer SyncService

Créer `src/app/core/services/sync.service.ts`:

```typescript
import { Injectable, NgZone } from '@angular/core';
import { BehaviorSubject, Observable, interval, fromEvent } from 'rxjs';
import { filter, switchMap } from 'rxjs/operators';
import { SyncOperation } from '../models/datamodel';
import { QueueService } from './queue.service';
import { FirestoreService } from './firestore.service';

@Injectable({
  providedIn: 'root'
})
export class SyncService {
  private isOnlineSubject = new BehaviorSubject<boolean>(navigator.onLine);
  isOnline$ = this.isOnlineSubject.asObservable();

  private isSyncingSubject = new BehaviorSubject<boolean>(false);
  isSyncing$ = this.isSyncingSubject.asObservable();

  private pendingCountSubject = new BehaviorSubject<number>(0);
  pendingCount$ = this.pendingCountSubject.asObservable();

  private syncErrorSubject = new BehaviorSubject<string | null>(null);
  syncError$ = this.syncErrorSubject.asObservable();

  constructor(
    private queue: QueueService,
    private firestore: FirestoreService,
    private ngZone: NgZone
  ) {
    this.setupConnectivityListener();
    this.setupAutoSync();
  }

  private setupConnectivityListener() {
    this.ngZone.runOutsideAngular(() => {
      fromEvent(window, 'online')
        .pipe(filter(() => true))
        .subscribe(() => {
          this.ngZone.run(() => {
            this.isOnlineSubject.next(true);
            this.syncError$.subscribe(() => this.startSync());
          });
        });

      fromEvent(window, 'offline')
        .pipe(filter(() => true))
        .subscribe(() => {
          this.ngZone.run(() => this.isOnlineSubject.next(false));
        });
    });
  }

  private setupAutoSync() {
    this.ngZone.runOutsideAngular(() => {
      interval(5000) // Sync every 5 seconds
        .pipe(
          switchMap(() => this.isOnline$),
          filter(isOnline => isOnline)
        )
        .subscribe(() => {
          this.ngZone.run(() => this.startSync());
        });
    });
  }

  async startSync() {
    if (this.isSyncingSubject.value) return;

    this.isSyncingSubject.next(true);
    try {
      const operations = await this.queue.getPendingOperations();
      this.pendingCountSubject.next(operations.length);

      // Group by matchId for batching
      const grouped = new Map<number, SyncOperation[]>();
      operations.forEach(op => {
        if (!grouped.has(op.matchId)) grouped.set(op.matchId, []);
        grouped.get(op.matchId)!.push(op);
      });

      // Sync each group
      for (const [matchId, ops] of grouped.entries()) {
        for (const op of ops) {
          await this.syncOperation(op);
        }
      }

      this.syncErrorSubject.next(null);
    } catch (error: any) {
      this.syncErrorSubject.next(error.message);
    } finally {
      this.isSyncingSubject.next(false);
      const pending = await this.queue.getPendingOperations();
      this.pendingCountSubject.next(pending.length);
    }
  }

  private async syncOperation(operation: SyncOperation) {
    await this.queue.updateOperationStatus(operation.id, 'syncing');

    try {
      const maxRetries = 5;
      const baseDelay = 1000; // 1 second

      for (let attempt = 0; attempt < maxRetries; attempt++) {
        try {
          if (operation.operation === 'create') {
            await this.firestore.createEvent(operation.data);
          } else if (operation.operation === 'update') {
            await this.firestore.updateEvent(operation.data);
          } else if (operation.operation === 'delete') {
            await this.firestore.deleteEvent(operation.data.id);
          }

          await this.queue.updateOperationStatus(operation.id, 'synced', 0);
          return;
        } catch (error: any) {
          if (attempt < maxRetries - 1 && error.code !== 'permission-denied') {
            const delay = baseDelay * Math.pow(2, attempt);
            await new Promise(resolve => setTimeout(resolve, delay));
          } else if (attempt === maxRetries - 1) {
            throw error;
          }
        }
      }
    } catch (error: any) {
      if (error.code === 'permission-denied') {
        await this.queue.updateOperationStatus(operation.id, 'conflict');
      } else {
        await this.queue.updateOperationStatus(
          operation.id,
          'failed',
          operation.retryCount + 1
        );
      }
      throw error;
    }
  }

  getOnlineStatus(): Observable<boolean> {
    return this.isOnline$;
  }

  getSyncingStatus(): Observable<boolean> {
    return this.isSyncing$;
  }

  getPendingCount(): Observable<number> {
    return this.pendingCount$;
  }

  getSyncError(): Observable<string | null> {
    return this.syncError$;
  }
}
```

### 3. Créer FirestoreService

Créer `src/app/core/services/firestore.service.ts`:

```typescript
import { Injectable } from '@angular/core';
import { collection, addDoc, updateDoc, deleteDoc, doc, writeBatch, Firestore } from 'firebase/firestore';
import { db } from '../config/firebase.config';
import { Evenement } from '../models/datamodel';

@Injectable({
  providedIn: 'root'
})
export class FirestoreService {
  constructor() {}

  async createEvent(event: Evenement): Promise<string> {
    try {
      const eventsCollection = collection(
        db,
        'teams',
        event.matchId.toString(),
        'matches',
        event.matchId.toString(),
        'evenements'
      );
      const docRef = await addDoc(eventsCollection, {
        ...event,
        syncedAt: new Date().toISOString()
      });
      return docRef.id;
    } catch (error) {
      console.error('Error creating event:', error);
      throw error;
    }
  }

  async updateEvent(event: Evenement): Promise<void> {
    try {
      const eventRef = doc(
        db,
        'teams',
        event.matchId.toString(),
        'matches',
        event.matchId.toString(),
        'evenements',
        event.id.toString()
      );
      await updateDoc(eventRef, {
        ...event,
        syncedAt: new Date().toISOString()
      });
    } catch (error) {
      console.error('Error updating event:', error);
      throw error;
    }
  }

  async deleteEvent(eventId: number): Promise<void> {
    try {
      const eventRef = doc(db, 'evenements', eventId.toString());
      await deleteDoc(eventRef);
    } catch (error) {
      console.error('Error deleting event:', error);
      throw error;
    }
  }

  async batchSync(operations: Array<{
    type: 'create' | 'update' | 'delete',
    data: any
  }>): Promise<void> {

    }

    await batch.commit();
  }
}
```
