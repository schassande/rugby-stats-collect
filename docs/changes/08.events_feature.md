# Phase 08 - Events Feature (Core)

## Objectif

Implémenter la collecte d'événements (le cœur du projet) avec validation métier cascade.

## Étapes

### 1. Créer EventService

Créer `src/app/features/events/services/event.service.ts`:

```typescript
import { Injectable } from '@angular/core';
import { BehaviorSubject, Observable } from 'rxjs';
import { map } from 'rxjs/operators';
import { Evenement, SyncOperation } from '@core/models/datamodel';
import { DatabaseService } from '@core/services/database.service';
import { QueueService } from '@core/services/queue.service';
import { v4 as uuid } from 'uuid';

@Injectable({
  providedIn: 'root'
})
export class EventService {
  private eventsSubject = new Map<number, BehaviorSubject<Evenement[]>>();

  constructor(
    private db: DatabaseService,
    private queue: QueueService
  ) {}

  private getEventsSubject(matchId: number): BehaviorSubject<Evenement[]> {
    if (!this.eventsSubject.has(matchId)) {
      this.eventsSubject.set(matchId, new BehaviorSubject<Evenement[]>([]));
      this.loadEvents(matchId);
    }
    return this.eventsSubject.get(matchId)!;
  }

  private async loadEvents(matchId: number) {
    const events = await this.db.getEventsByMatch(matchId);
    this.getEventsSubject(matchId).next(events);
  }

  async addEvent(matchId: number, event: Partial<Evenement>): Promise<Evenement> {
    // Valider métier
    const errors = this.validateEventData(event);
    if (errors.length > 0) throw new Error(errors.join(', '));

    // Créer événement
    const newEvent: Omit<Evenement, 'id'> = {
      matchId,
      ...event as any,
      createdAt: new Date().toISOString(),
      syncedAt: undefined
    };

    // Ajouter à DB
    const dbEvent = await this.db.addEvent(newEvent);

    // Ajouter à queue
    const operation: SyncOperation = {
      id: uuid(),
      evenementId: dbEvent.id,
      matchId,
      operation: 'create',
      status: 'pending',
      data: dbEvent,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      retryCount: 0
    };
    await this.queue.addOperation(operation);

    // Émettre
    const current = this.getEventsSubject(matchId).value;
    this.getEventsSubject(matchId).next([...current, dbEvent]);

    return dbEvent;
  }

  async deleteEvent(matchId: number, eventId: number): Promise<void> {
    await this.db.deleteEvent(eventId);

    const operation: SyncOperation = {
      id: uuid(),
      evenementId: eventId,
      matchId,
      operation: 'delete',
      status: 'pending',
      data: {} as Evenement,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      retryCount: 0
    };
    await this.queue.addOperation(operation);

    const current = this.getEventsSubject(matchId).value.filter(e => e.id !== eventId);
    this.getEventsSubject(matchId).next(current);
  }

  getEventsByMatch(matchId: number): Observable<Evenement[]> {
    return this.getEventsSubject(matchId).asObservable();
  }

  getEventsByMatchAntiChronological(matchId: number): Observable<Evenement[]> {
    return this.getEventsSubject(matchId).pipe(
      map(events => [...events].sort((a, b) => 
        new Date(b.instant).getTime() - new Date(a.instant).getTime()
      ))
    );
  }

  validateEventData(event: Partial<Evenement>): string[] {
    const errors: string[] = [];

    if (!event.nature) errors.push('Nature requis');
    if (!event.type) errors.push('Type requis');
    if (!event.equipe) errors.push('Équipe requis');
    if (!event.resultat) errors.push('Résultat requis');

    // Valider résultat pour le type
    const validResults = this.getValidResultsForType(event.type);
    if (event.resultat && !validResults.includes(event.resultat)) {
      errors.push(`Résultat invalide pour ${event.type}`);
    }

    return errors;
  }

  getValidResultsForType(type: any): string[] {
    const validMap: any = {
      'ESSAI': ['MARQUE', 'RATE'],
      'TRANSFORMATION': ['REUSSITE', 'ECHEC'],
      'PENALITE': ['MARQUE', 'RATE'],
      'DROP': ['MARQUE', 'RATE'],
      'TOUCHE': ['GAGNEE', 'PERDUE', 'CONTRE', 'PENALISEE'],
      'MELEE': ['GAGNEE', 'PENALISEE', 'SUBIE', 'CONTRE'],
      'COUP_ENVOI': ['RECUPERE', 'NON_RECUPERE'],
      'RENVOI': ['RECUPERE', 'NON_RECUPERE'],
      'EN_AVANT': ['PENALISEE'],
      'PENALITE_COMMIS': ['PENALISEE'],
      'COUP_DE_PIED_FRANC': ['PENALISEE'],
      'TOUCHE_TROUVEE': ['NEUTRE'],
      'TOUCHE_NON_TROUVEE': ['NEUTRE'],
      'ESSAI_SAUVE_SUR_MAUL': ['GAGNEE']
    };
    return validMap[type] || [];
  }

  getSubtypesForNature(nature: string): string[] {
    const map: any = {
      'SCORE': ['ESSAI', 'TRANSFORMATION', 'PENALITE', 'DROP'],
      'CONQUETE': ['TOUCHE', 'MELEE', 'COUP_ENVOI', 'RENVOI'],
      'DISCIPLINE': ['EN_AVANT', 'PENALITE_COMMIS', 'COUP_DE_PIED_FRANC'],
      'INDICATEUR': ['TOUCHE_TROUVEE', 'TOUCHE_NON_TROUVEE', 'ESSAI_SAUVE_SUR_MAUL']
    };
    return map[nature] || [];
  }
}
```

### 2. Créer EventFormComponent

Créer `src/app/features/events/pages/event-form/event-form.component.ts`:

```typescript
import { Component, OnInit } from '@angular/core';
import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { ActivatedRoute, Router } from '@angular/router';
import { CommonModule } from '@angular/common';
import { DialogModule } from 'primeng/dialog';
import { ButtonModule } from 'primeng/button';
import { ButtonGroupModule } from 'primeng/buttongroup';
import { DropdownModule } from 'primeng/dropdown';
import { InputMaskModule } from 'primeng/inputmask';
import { InputTextareaModule } from 'primeng/inputtextarea';
import { ConfirmDialogModule } from 'primeng/confirmdialog';
import { MessageModule } from 'primeng/message';
import { EventService } from '../../services/event.service';
import { ConfirmationService } from 'primeng/api';

@Component({
  selector: 'app-event-form',
  standalone: true,
  imports: [
    CommonModule, ReactiveFormsModule, DialogModule, ButtonModule,
    ButtonGroupModule, DropdownModule, InputMaskModule, InputTextareaModule,
    ConfirmDialogModule, MessageModule
  ],
  providers: [ConfirmationService],
  template: `
    <p-dialog 
      [(visible)]="displayDialog"
      [modal]="true"
      [header]="'Événement'"
      (onHide)="cancel()">

      <!-- Context -->
      <div class="match-context">
        {{ homeTeam }} {{ score.nous }} - {{ score.adversaire }} {{ awayTeam }}
        <br>
        1er mi-temps · {{ currentTime }}
      </div>

      <form [formGroup]="form" class="event-form">
        <!-- Nature -->
        <label>Type d'événement</label>
        <p-buttonGroup formControlName="nature">
          <p-button 
            *ngFor="let nat of natures"
            [label]="nat"
            [ngClass]="{ 'p-button-success': form.get('nature')?.value === nat }"
            (click)="selectNature(nat)">
          </p-button>
        </p-buttonGroup>

        <!-- Sous-type (dynamique) -->
        <label>Sous-type</label>
        <p-dropdown 
          formControlName="type"
          [options]="subtypeOptions"
          placeholder="Sélectionnez..."
          (onChange)="updateResults()">
        </p-dropdown>

        <!-- Équipe -->
        <label>Équipe</label>
        <p-buttonGroup formControlName="equipe">
          <p-button label="NOUS" value="NOUS"></p-button>
          <p-button label="ADV" value="ADV"></p-button>
        </p-buttonGroup>

        <!-- Résultat (dynamique) -->
        <label>Résultat</label>
        <p-dropdown 
          formControlName="resultat"
          [options]="resultOptions"
          placeholder="Sélectionnez...">
        </p-dropdown>

        <!-- Minute -->
        <label>Minute (optionnel)</label>
        <p-inputMask 
          formControlName="minute"
          mask="99:99"
          placeholder="mm:ss">
        </p-inputMask>

        <!-- Commentaire -->
        <label>Commentaire (optionnel)</label>
        <textarea 
          pInputTextarea
          formControlName="commentaire"
          rows="3"
          placeholder="Notes...">
        </textarea>

        <!-- Erreurs -->
        <p-message 
          *ngIf="validationErrors.length > 0"
          severity="error"
          [text]="validationErrors.join(', ')">
        </p-message>
      </form>

      <ng-template pTemplate="footer">
        <p-button 
          label="Annuler"
          (click)="cancel()"
          styleClass="p-button-text">
        </p-button>
        <p-button 
          *ngIf="isEdit"
          label="Supprimer"
          icon="pi pi-trash"
          severity="danger"
          (click)="delete()">
        </p-button>
        <p-button 
          label="Enregistrer"
          icon="pi pi-check"
          (click)="save()"
          [loading]="isSaving">
        </p-button>
      </ng-template>
    </p-dialog>

    <p-confirmDialog></p-confirmDialog>
  `,
  styleUrls: ['./event-form.component.scss']
})
export class EventFormComponent implements OnInit {
  form: FormGroup;
  displayDialog = true;
  isEdit = false;
  isSaving = false;
  validationErrors: string[] = [];

  natures = ['SCORE', 'CONQUETE', 'DISCIPLINE', 'INDICATEUR'];
  subtypeOptions: string[] = [];
  resultOptions: string[] = [];

  homeTeam = 'COXS';
  awayTeam = 'ADV';
  score = { nous: 0, adversaire: 0 };
  currentTime = '00:00';
  matchId = 0;

  constructor(
    private fb: FormBuilder,
    private eventService: EventService,
    private route: ActivatedRoute,
    private router: Router,
    private confirmationService: ConfirmationService
  ) {
    this.form = this.fb.group({
      nature: ['SCORE', Validators.required],
      type: ['', Validators.required],
      equipe: ['NOUS', Validators.required],
      resultat: ['', Validators.required],
      minute: [''],
      commentaire: ['']
    });
  }

  ngOnInit() {
    const matchIdParam = this.route.snapshot.queryParamMap.get('matchId');
    if (matchIdParam) {
      this.matchId = +matchIdParam;
    }
    this.updateSubtypes();
  }

  selectNature(nature: string) {
    this.form.patchValue({ nature, type: '', resultat: '' });
    this.updateSubtypes();
    this.validationErrors = [];
  }

  updateSubtypes() {
    const nature = this.form.get('nature')?.value;
    this.subtypeOptions = this.eventService.getSubtypesForNature(nature);
    this.form.patchValue({ type: '', resultat: '' });
    this.updateResults();
  }

  updateResults() {
    const type = this.form.get('type')?.value;
    this.resultOptions = this.eventService.getValidResultsForType(type);
    this.form.patchValue({ resultat: '' });
  }

  async save() {
    this.validationErrors = this.eventService.validateEventData(this.form.value);
    if (this.validationErrors.length > 0) return;

    this.isSaving = true;
    try {
      await this.eventService.addEvent(this.matchId, this.form.value);
      this.router.navigate(['/app/matches', this.matchId]);
    } catch (error: any) {
      this.validationErrors = [error.message];
    } finally {
      this.isSaving = false;
    }
  }

  delete() {
    this.confirmationService.confirm({
      message: 'Supprimer cet événement?',
      header: 'Confirmation',
      icon: 'pi pi-exclamation-triangle',
      accept: async () => {
        // Implémenter suppression
      }
    });
  }

  cancel() {
    this.router.back();
  }
}
```
